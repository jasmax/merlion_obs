Using it

# before you push: run the dataset, diff against main, gate
python -m merlion_devobs.cli run --config configs/main.json -m "router: few-shot" --baseline main

# what changed, case by case, and where it diverged
python -m merlion_devobs.cli diff r05 r06
python -m merlion_devobs.cli show r06 E0067
python -m merlion_devobs.cli clusters r06

# grow the dataset from production: CS overrides need a label
python -m merlion_devobs.cli promote ../out/traces.db
